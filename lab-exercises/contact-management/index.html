phpMyAdmin Database & PHP Backend Code

1. phpMyAdmin SQL Table Creation (contacts_db.sql)

CREATE DATABASE contacts_db;
USE contacts_db;

CREATE TABLE contacts (
    id INT AUTO_INCREMENT PRIMARY KEY,
    name VARCHAR(100) NOT NULL,
    phone VARCHAR(20) NOT NULL,
    email VARCHAR(100) NOT NULL,
    category ENUM('Work', 'Personal', 'Family', 'Friends', 'Other') DEFAULT 'Personal',
    notes TEXT,
    created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP
);

2. PHP REST API for CRUD (api.php)

<?php
$conn = new mysqli("localhost", "root", "", "contacts_db");
$method = $_SERVER['REQUEST_METHOD'];

if ($method === 'GET') {
    // VIEW CONTACTS
    $res = $conn->query("SELECT * FROM contacts ORDER BY id DESC");
    echo json_encode($res->fetch_all(MYSQLI_ASSOC));
} elseif ($method === 'POST') {
    // ADD CONTACT
    $stmt = $conn->prepare("INSERT INTO contacts (name, phone, email, category, notes) VALUES (?, ?, ?, ?, ?)");
    $stmt->bind_param("sssss", $_POST['name'], $_POST['phone'], $_POST['email'], $_POST['category'], $_POST['notes']);
    $stmt->execute();
} elseif ($method === 'PUT') {
    // EDIT CONTACT
    $data = json_decode(file_get_contents("php://input"), true);
    $stmt = $conn->prepare("UPDATE contacts SET name=?, phone=?, email=?, category=?, notes=? WHERE id=?");
    $stmt->bind_param("sssssi", $data['name'], $data['phone'], $data['email'], $data['category'], $data['notes'], $data['id']);
    $stmt->execute();
} elseif ($method === 'DELETE') {
    // DELETE CONTACT
    $stmt = $conn->prepare("DELETE FROM contacts WHERE id=?");
    $stmt->bind_param("i", $_GET['id']);
    $stmt->execute();
}
?>
Operation successful!
