jQuery $.ajax Implementation (Exercise 10)

Lab Exercise Problem Statement Snippet:

// When the "Load Books" button is clicked, an Ajax GET request is made
// using jQuery's $.ajax method to retrieve a list of books from
// a hypothetical API endpoint (https://api.example.com/books).
// The JSON response is then parsed, and a list of books
// (including titles and authors) is displayed on the webpage.

$("#loadBooksBtn").on("click", function() {
    $.ajax({
        url: "https://api.example.com/books", // Or actual API endpoint
        type: "GET",
        dataType: "json",
        beforeSend: function() {
            $("#loaderBox").show();
        },
        success: function(response) {
            $("#loaderBox").hide();
            // Parse JSON and render titles & authors
            response.forEach(function(book) {
                $("#booksList").append(
                    "<li><strong>" + book.title + "</strong> by " + book.author + "</li>"
                );
            });
        },
        error: function(xhr, status, error) {
            $("#loaderBox").hide();
            alert("Error loading books: " + error);
        }
    });
});
