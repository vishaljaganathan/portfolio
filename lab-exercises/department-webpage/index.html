Department Webpage Backend Code (PHP & phpMyAdmin)

1. phpMyAdmin Database Schema (department_db.sql)

CREATE DATABASE department_db;
USE department_db;

CREATE TABLE students (
    id INT AUTO_INCREMENT PRIMARY KEY,
    reg_no VARCHAR(50) NOT NULL UNIQUE,
    full_name VARCHAR(100) NOT NULL,
    email VARCHAR(100) NOT NULL UNIQUE,
    program VARCHAR(50) NOT NULL,
    semester VARCHAR(20) NOT NULL,
    password_hash VARCHAR(255) NOT NULL,
    registered_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP
);

2. PHP Registration Handler (register.php)

<?php
require_once "db.php";
$regNo    = $_POST['reg_no'];
$fullName = $_POST['full_name'];
$email    = $_POST['email'];
$program  = $_POST['program'];
$semester = $_POST['semester'];
$password = password_hash($_POST['password'], PASSWORD_DEFAULT);

$stmt = $conn->prepare("INSERT INTO students (reg_no, full_name, email, program, semester, password_hash) VALUES (?, ?, ?, ?, ?, ?)");
$stmt->bind_param("ssssss", $regNo, $fullName, $email, $program, $semester, $password);
$stmt->execute();
echo json_encode(["status" => "success", "message" => "Registration saved to phpMyAdmin!"]);
?>

3. PHP Login Handler (login.php)

<?php
session_start();
require_once "db.php";
$identifier = $_POST['identifier'];
$password   = $_POST['password'];

$stmt = $conn->prepare("SELECT * FROM students WHERE reg_no = ? OR email = ?");
$stmt->bind_param("ss", $identifier, $identifier);
$stmt->execute();
$user = $stmt->get_result()->fetch_assoc();

if ($user && password_verify($password, $user['password_hash'])) {
    $_SESSION['user'] = $user;
    echo json_encode(["status" => "success", "user" => $user]);
} else {
    echo json_encode(["status" => "error", "message" => "Invalid credentials!"]);
}
?>
